\section{Formal Verification Methodology}
\label{sec:formal-verification}

This section adapts the search algorithm for the maximal certifiable sublevel set $\mathcal{V}_\rho$ 
proposed in~\cite{yangLyapunovstableNeuralControl2024}. 
Specifically, for a given target level $\rho > 0$, formal verification aims to certify that all states within 
the candidate region $\mathcal{V}_\rho := \{ x \in \mathcal{X}_C \mid V_\phi(x) \le \rho \}$ satisfy controlled 
regional invariance and exponential decay:
\begin{equation}
    \label{eq:lyapunov-certified-property}
    V_\phi(x) \le \rho \implies 
    \begin{cases} 
        x^+ \in \mathcal{X}_C, \\ 
        V_\phi(x^+) \le (1 - \kappa) V_\phi(x),
    \end{cases}
\end{equation}
where $\kappa \in (0, 1)$ specifies the targeted convergence rate and $x^+ = f(x, \pi_\theta(x))$ denotes the discrete 
one-step state transition.

Before evaluating the stability conditions, the equilibrium $x^\star$ must be explicitly excluded from the bounded 
certification domain 
$\mathcal{X}_C := \{ x \in \mathbb{R}^n \mid \underline{x}_C \le x \le \overline{x}_C \} \subseteq \mathcal{X}_V$. 
By construction, $V_\phi(x^\star) = 0$ and $x^+ = x^\star$ hold at the equilibrium, 
causing the strict positivity and strict decrease conditions to evaluate to false at this exact point. 
Since these conditions are only required to hold for $x \neq x^\star$, and positive definiteness is structurally 
enforced by the network architecture, formal verification is instead performed over the punctured evaluation domain 
$\mathcal{X}_{\mathrm{eval}} := \mathcal{X}_C \setminus \mathcal{X}_{\mathrm{hole}}$, where 
$\mathcal{X}_{\mathrm{hole}}$ denotes a small hyper-rectangular region centered at $x^\star$. 
For the solver to handle the punctured certification domain, $\mathcal{X}_{\mathrm{eval}}$ is split into multiple subregions, 
as shown in \cref{fig:hole-exclusion}.

To certify property \labelcref{eq:lyapunov-certified-property} over $\mathcal{X}_{\mathrm{eval}}$ using 
$\alpha\beta$-CROWN~\cite{wang2021beta,xuFastCompleteEnabling2021}, 
the implication is rewritten into a disjunctive formulation supported by the verifier: 
a state must either lie outside $\mathcal{V}_\rho$ or satisfy all stability conditions inside 
$\mathcal{X}_{\mathrm{eval}}$. The individual logical terms are defined as follows:
\begin{subequations}\label{eq:certification-specification}
    \begin{align}
        \Phi_{\mathrm{sub}}
            &\equiv V_\phi(x) > \rho,
            \label{eq:outside-certification}\\
        \Phi_{\mathrm{pos}}
            &\equiv V_\phi(x) > 0,
            \label{eq:positive-certification}\\
        \Phi_{\mathrm{inv}}
            &\equiv \underline{x}_C \le x^+ \le \overline{x}_C,
            \label{eq:invariance-certification}\\
        \Phi_{\mathrm{dec}}
            &\equiv (1 - \kappa) V_\phi(x) > V_\phi(x^+).
            \label{eq:decrease-certification}
    \end{align}
\end{subequations}
Here, \labelcref{eq:outside-certification} excludes states outside the target sublevel set, while 
\labelcref{eq:positive-certification,eq:invariance-certification,eq:decrease-certification} explicitly enforce positive definiteness, 
state constraint invariance, and exponential decrease. Combining these terms yields the overall state-wise specification:
\begin{equation}
    \label{eq:total-certification-formulation}
    \Phi(x; \rho) \equiv \Phi_{\mathrm{sub}} \lor \left( 
        \Phi_{\mathrm{pos}} 
        \land \Phi_{\mathrm{inv}} 
        \land \Phi_{\mathrm{dec}} 
    \right).
\end{equation}

To determine whether a candidate level $\rho$ is valid across the entire continuous evaluation domain, 
the global verification condition $\Psi(\rho)$ evaluates whether $\Phi(x; \rho)$ holds for all states:
\begin{equation}
    \label{eq:domain-certification}
    \Psi(\rho) \equiv \forall x \in \mathcal{X}_{\mathrm{eval}} : \Phi(x; \rho).
\end{equation}

The $\alpha\beta$-CROWN verifier evaluates $\Psi(\rho)$ iteratively within the scaling and bisection procedure 
outlined in \cref{alg:search-and-bisect}.
The search operates in two distinct phases: 
First, a scaling phase exponentially expands or contracts $\rho$ by a factor $q > 1$ to bracket the valid sublevel set 
boundary within an interval $[\rho_{\mathrm{lo}}, \rho_{\mathrm{up}}]$.
Second, a bisection phase narrows this interval until the gap falls below a tolerance $\varepsilon$. 
Upon termination, the lower bound yields the maximum certified value $\rho^\star \approx \rho_{\mathrm{lo}}$.
Compared to the proposed formulation by Yang in~\cite{yangLyapunovstableNeuralControl2024}, 
explicit iteration limits $N_{\mathrm{scale}}$ and $N_{\mathrm{bisect}}$ are integrated into both phases. 
This caps the maximum number of verifier calls, effectively preventing excessive computational 
overhead on difficult verification problems.

\begin{algorithm}[htpb]
    \caption{Certified Value Search via Scaling and Bisection}
    \label{alg:search-and-bisect}
    \SetAlgoLined
    \KwIn{$\rho_0, \rho_{\min}$, evaluator $\Psi$, scale $q > 1$, tol $\varepsilon$, limits $N_{\mathrm{scale}}, N_{\mathrm{bisect}}$}
    \KwOut{Max certified value $\rho^\star$}

    \BlankLine
    $\rho \leftarrow \max(\rho_0, \rho_{\min})$\;

    \BlankLine
    \tcp{Phase 1: Scaling}
    \eIf{$\Psi(\rho)$}{
        \For{$k \leftarrow 1$ \KwTo $N_{\mathrm{scale}}$}{
            \If{$\neg \Psi(\rho)$}{\textbf{break}\;}
            $\rho_{\mathrm{lo}} \leftarrow \rho;\quad \rho \leftarrow \rho \cdot q$\;
        }
        $\rho_{\mathrm{up}} \leftarrow \rho$\;
    }{
        \For{$k \leftarrow 1$ \KwTo $N_{\mathrm{scale}}$}{
            \If{$\Psi(\rho) \lor \rho = \rho_{\min}$}{\textbf{break}\;}
            $\rho_{\mathrm{up}} \leftarrow \rho;\quad \rho \leftarrow \max(\rho_{\min}, \rho / q)$\;
        }
        \If{$\neg \Psi(\rho)$}{\KwRet $0$\ \tcp*{No valid value $\ge \rho_{\min}$}}
        $\rho_{\mathrm{lo}} \leftarrow \rho$\;
    }

    \BlankLine
    \tcp{Phase 2: Bisection}
    \For{$k \leftarrow 1$ \KwTo $N_{\mathrm{bisect}}$}{
        \If{$\rho_{\mathrm{up}} - \rho_{\mathrm{lo}} \le \varepsilon$}{\textbf{break}\;}
        $\rho_{\mathrm{mid}} \leftarrow \frac{1}{2}(\rho_{\mathrm{lo}} + \rho_{\mathrm{up}})$\;
        \eIf{$\Psi(\rho_{\mathrm{mid}})$}{
            $\rho_{\mathrm{lo}} \leftarrow \rho_{\mathrm{mid}}$\;
        }{
            $\rho_{\mathrm{up}} \leftarrow \rho_{\mathrm{mid}}$\;
        }
    }

    \BlankLine
    \KwRet $\rho_{\mathrm{lo}}$\;
\end{algorithm}
