\section{Formal Verification Methodology}
\label{sec:formal-verification}

This section adapts the search algorithm for the maximal certifiable sublevel set $\mathcal{V}_\rho$ 
proposed in \cite{yangLyapunovstableNeuralControl2024}. 
However, iteration limits for scaling and bisection are integrated into \cref{alg:search-and-bisect} 
to reduce computational overhead during evaluation.


\begin{algorithm}[htpb]
    \caption{Certified Value Search via Scaling and Bisection}
    \label{alg:search-and-bisect}
    \SetAlgoLined
    \KwIn{$\rho_0, \rho_{\min}$, evaluator $\Phi$, scale $s > 1$, tol $\varepsilon$, limits $N_{\text{scale}}, N_{\text{bisect}}$}
    \KwOut{Max certified value $\rho^*$}

    \BlankLine
    $\rho \leftarrow \max(\rho_0, \rho_{\min})$\;

    \BlankLine
    \tcp{Phase 1: Scaling}
    \eIf{$\Phi(\rho)$}{
        \For{$k \leftarrow 1$ \KwTo $N_{\text{scale}}$}{
            \If{$\neg \Phi(\rho)$}{\textbf{break}\;}
            $\rho_{\text{lo}} \leftarrow \rho;\quad \rho \leftarrow \rho \cdot s$\;
        }
        $\rho_{\text{up}} \leftarrow \rho$\;
    }{
        \For{$k \leftarrow 1$ \KwTo $N_{\text{scale}}$}{
            \If{$\Phi(\rho) \lor \rho = \rho_{\min}$}{\textbf{break}\;}
            $\rho_{\text{up}} \leftarrow \rho;\quad \rho \leftarrow \max(\rho_{\min}, \rho / s)$\;
        }
        \If{$\neg \Phi(\rho)$}{\KwRet $0$\ \tcp*{No valid value $\ge \rho_{\min}$}}
        $\rho_{\text{lo}} \leftarrow \rho$\;
    }

    \BlankLine
    \tcp{Phase 2: Bisection}
    \For{$k \leftarrow 1$ \KwTo $N_{\text{bisect}}$}{
        \If{$\rho_{\text{up}} - \rho_{\text{lo}} \le \varepsilon$}{\textbf{break}\;}
        $\rho_{\text{mid}} \leftarrow \frac{1}{2}(\rho_{\text{lo}} + \rho_{\text{up}})$\;
        \eIf{$\Phi(\rho_{\text{mid}})$}{
            $\rho_{\text{lo}} \leftarrow \rho_{\text{mid}}$\;
        }{
            $\rho_{\text{up}} \leftarrow \rho_{\text{mid}}$\;
        }
    }

    \BlankLine
    \KwRet $\rho_{\text{lo}}$\;
\end{algorithm}

To formally certify the Lyapunov candidate, the $\alpha\beta$-CROWN verifier evaluates a 
logical formulation $\Phi(x; \rho)$ over a bounded certification domain 
$\mathcal{B}_C = \{ x \in \mathbb{R}^n \mid \underline{x}_C \le x \le \bar{x}_C \} \subseteq \mathcal{B}_V$. 
The goal is to verify that all states in $\mathcal{V}_\rho \cap \mathcal{B}_C$ satisfy the required stability conditions.
This implication is rewritten into a disjunctive form that is directly supported by the verifier: 
a state must either lie outside $\mathcal{V}_\rho$ or satisfy all stability conditions.

The individual logical terms from which this specification is composed are defined as follows:
\begin{subequations}\label{eq:certification-specification}
    \begin{align}
        \Phi_{\mathrm{sub}}
            &\equiv V_\theta(x) > \rho,
            \label{eq:outside-certification}\\
        \Phi_{\mathrm{pos}}
            &\equiv V_\theta(x) > 0,
            \label{eq:positive-certification}\\
        \Phi_{\mathrm{inv}}
            &\equiv \underline{x}_C \le x^+ \le \bar{x}_C,
            \label{eq:invariance-certification}\\
        \Phi_{\mathrm{dec}}
            &\equiv (1 - \kappa) V_\theta(x) - V_\theta(x^+) > 0,
            \label{eq:decrease-certification}
    \end{align}
\end{subequations}
where $x^+ = f(x, \pi_\theta(x))$ denotes the discrete one-step state transition. Here, \cref{eq:outside-certification} 
determines whether $x$ lies outside the target sublevel set $\mathcal{V}_\rho$. For states within $\mathcal{V}_\rho$, 
\cref{eq:positive-certification} enforces strict positive definiteness away from $x^*$, \cref{eq:invariance-certification} 
guarantees forward invariance within $\mathcal{B}_C$, and \cref{eq:decrease-certification} ensures the strict Lyapunov decrease property.

Finally, combining the sublevel exclusion with the joint stability conditions yields the total specification evaluated by the verifier:
\begin{equation}
    \label{eq:total-certification-formulation}
    \Phi(x; \rho) \equiv \Phi_{\mathrm{sub}} \lor \left( 
        \Phi_{\mathrm{pos}} 
        \land \Phi_{\mathrm{inv}} 
        \land \Phi_{\mathrm{dec}} 
    \right).
\end{equation}


The equilibrium $x^*$ must be excluded from the verification domain. 
By construction, $V_\theta(x^*) = 0$ and $x^+ = x^*$ hold at the equilibrium, 
causing the strict positivity ($\Phi_{\mathrm{pos}}$) and strict decrease ($\Phi_{\mathrm{dec}}$) 
conditions to evaluate to false at $x^*$. Since these conditions hold only for $x \neq x^*$ 
and positive definiteness is enforced by construction, formal verification is performed over 
the punctured domain $\mathcal{B}_C \setminus \mathcal{B}_{\mathrm{hole}}$, 
where $\mathcal{B}_{\mathrm{hole}}$ denotes a small hyper-rectangular region 
centered at $x^*$, as shown in \cref{fig:hole-exclusion}.
