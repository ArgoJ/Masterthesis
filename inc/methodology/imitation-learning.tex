\section{Neural Policy Approximation and Imitation Learning}
\label{sec:imitation-learning}


\paragraph{Data Generation} uses the MPC expert policy $\pi_0$ defined in \cref{subsec:closed-loop-stability}.
To ensure solver convergence and obtain consistent expert trajectories during data generation, 
the shifted initialization is used~\citeSec{12.5}{gruneNonlinearModelPredictive2017}. 
By shifting the previously computed optimal sequence $(u_{i|k}^\star)_{i=0}^{N-1}$ and appending the local terminal controller 
$\pi_f(x_{N|k}^\star)$, the optimization problem initializes from a feasible candidate trajectory. 
This prevents the solver from getting stuck in local minima between adjacent sampling instances.
Closed-loop trajectories $x_{k+1} = f(x_k, \pi_0(x_k))$ are initialized from states sampled across the mpc sampling
set $\mathcal{X}_{\mathrm{MPC}} := \left\{ x \in \mathcal{X} \mid \underline{x}_{\mathrm{MPC}} \leq x \leq \overline{x}_{\mathrm{MPC}} \right\}$.
However, the training set is defined as $\mathcal{X}_\pi := \left\{ x \in \mathcal{X} \mid \underline{x}_\pi \leq x \leq \overline{x}_\pi \right\}$. 
Storing the visited states and their corresponding optimal control actions yields the dataset
\begin{equation}
    \label{eq:dataset-definition}
    \mathcal{D}_\pi := \left\{ (x^{(i)}, u^{(i)}) \right\}_{i=1}^{N_{\mathcal{D}}},
\end{equation}
where $x^{(i)} \in \mathcal{X}_\pi$ and $u^{(i)} = \pi_0(x^{(i)})$.


\paragraph{Imitation Policy} follows the structure proposed in~\cite{yangLyapunovstableNeuralControl2024}, 
as it enforces true control inputs at the equilibrium $\pi_\theta(x^\star) = u^\star$ and 
satisfies the control input constraints $\underline{u} \leq \pi_\theta(x) \leq \overline{u}$ 
by construction. The unconstrained neural policy $\tilde{\pi}_\theta$ is formulated as 
\begin{equation}
    \label{eq:unc-policy-arch}
    \tilde{\pi}_\theta(x) = h_\theta(x) - h_\theta(x^\star) + u^\star.
\end{equation}
where $h_\theta : \mathbb{R}^{n_x} \to \mathbb{R}^{n_u}$ is a neural network and 
$u^\star \in \mathcal{U} $ represents the desired control action at the equilibrium $x^\star$.
However, to ensure the satisfaction of the control input constraints, the unconstrained output $\tilde{\pi}_\theta(x)$ 
is clipped element-wise using $\clamp(\cdot, \underline{\cdot}, \overline{\cdot})$. 
This yields the neural policy visualized in \cref{fig:imitation-policy}:
\begin{equation}
    \label{eq:policy-arch}
    \pi_\theta(x) = \clamp(\tilde{\pi}_\theta(x), \underline{u}, \overline{u}).
\end{equation}

\paragraph{Imitation Objective} is a multi-objective loss, which consists of two parts: a state-weighted behavioral cloning loss 
and a dynamics-aware penalty. 
To account for different physical units and numerical scales across the state and control vectors, 
the loss terms are normalized using strictly positive diagonal scaling matrices
\begin{equation}
    S_x = \diag(x_{\mathrm{scale}}) \in \mathbb{R}^{n_x \times n_x}
    \quad \text{and} \quad
    S_u = \diag(u_{\mathrm{scale}}) \in \mathbb{R}^{n_u \times n_u}.
\end{equation}
Using $S_x$, the normalized distance of a state $x$ from the equilibrium $x^\star$ is quantified with the $\ell_1$-norm as
\begin{equation}
    \label{eq:state-distance}
    d(x) = \frac{1}{n_x} \left\| S_x^{-1}(x - x^\star) \right\|_1.
\end{equation}

To assign appropriate priority during behavioral cloning, a state-dependent weight is defined as
\begin{equation}
    \label{eq:state-weight}
    w_x(x) = w_{\min} + (1 - w_{\min}) \left( \frac{1}{1 + d(x)} \right)^{\gamma_x},
\end{equation}
where $w_{\min} \in (0, 1)$ is the baseline weight for states far from the
equilibrium and $\gamma_x \geq 0$ controls the decay rate.

\paragraph{State-Weighted Behavioral Cloning Loss} penalizes the squared normalized 
Euclidean distance between the neural network policy and the expert targets. 
For a mini-batch $\mathcal{B}_\pi \subset \mathcal{D}_\pi$ of size 
$|\mathcal{B}_\pi|$, the behavioral cloning loss is defined as
\begin{equation}
    \label{eq:state-weighted-loss}
    \loss_{\mathrm{BC}} = \frac{1}{|\mathcal{B}_\pi| n_u} 
    \sum_{(x, u) \in \mathcal{B}_\pi} w_x(x) \left\| 
        S_u^{-1} \big( \pi_\theta(x) - u \big) 
    \right\|_2^2.
\end{equation}
Here, the state-dependent weight $w_x(x)$ prioritizes approximation accuracy near the equilibrium.


\paragraph{Dynamics-Aware Constraint Loss} applies a one-step penalty for successor states that leave 
the admissible state set $\mathcal{X}_\pi$. To maintain non-zero gradient flow even in regions of the 
state space where the control input saturates, the one-step forward dynamics are evaluated for all 
$x \in \mathcal{B}_\pi$ using the unconstrained neural policy as $x^+ = f(x,\tilde{\pi}_\theta(x))$. 
Using the element-wise ReLU function defined in \labelcref{eq:relu}, the loss is formulated as
\begin{equation}
    \label{eq:dynamics-aware-loss}
    \loss_f = \frac{1}{|\mathcal{B}_\pi| n_x} \sum_{x \in \mathcal{B}_\pi} \left( 
        \left\| 
            \big[ x^+ - \overline{x}_\pi \big]_+ 
        \right\|_1 + \left\| 
            \big[ \underline{x}_\pi - x^+ \big]_+ 
        \right\|_1 
    \right).
\end{equation}
Here, a one-step evaluation outside the admissible set results in a penalty for the current state, 
forcing the learned policy to stay inside the admissible set. 


\paragraph{Total Imitation Loss} is the combination of the state-weighted behavioral cloning 
and dynamics-aware constraint losses resulting in a scalar objective. 
Scaling each term by its respective weight, $\lambda_{\mathrm{BC}}$ and $\lambda_f$, 
yields the overall policy loss:
\begin{equation}
    \label{eq:combined-policy-loss}
    \loss_\pi = 
        \lambda_{\mathrm{BC}} \loss_{\mathrm{BC}} + 
        \lambda_f \loss_f.
\end{equation}

By minimizing $\loss_\pi$, the policy $\pi_\theta$ learns to mimic the expert policy $\pi_0$, 
while penalizing state constraint violations in $\mathcal{X}_\pi$. 
In combination with the policy architecture in \labelcref{eq:policy-arch}, $\pi_\theta$ satisfies the input constraints 
$\underline{u} \leq \pi_\theta(x) \leq \overline{u}$ for all states $x \in \mathbb{R}^{n_x}$ and guarantees exact 
equilibrium control $\pi_\theta(x^\star) = u^\star$. 
Additionally, the state-dependent weighting in \labelcref{eq:state-weight} prioritizes approximation accuracy 
near the equilibrium, which may be beneficial for establishing Lyapunov stability.
Now, this imitation learning pipeline establishes an accurate expert imitation, 
serving as an initial policy for the subsequent Lyapunov co-training.
