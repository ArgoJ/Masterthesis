\section{Neural Policy Approximation and Imitation Learning}
\label{sec:imitation-learning}


\paragraph{Data Generation} uses the MPC expert policy $\pi_0$ defined in \cref{subsec:closed-loop-stability}.
Closed-loop trajectories $x_{k+1} = f_d(x_k, \pi_0(x_k))$ are initialized from states sampled across 
the training set $\mathcal{X}_\pi \subseteq \mathcal{X}$. 
Storing the visited states and their corresponding optimal control actions yields the dataset
\begin{equation}
    \label{eq:dataset-definition}
    \mathcal{D}_\pi = \left\{ (x^{(i)}, u^{(i)}) \right\}_{i=1}^{N_{\mathcal{D}}},
\end{equation}
where $x^{(i)} \in \mathcal{X}_\pi$ and $u^{(i)} = \pi_0(x^{(i)})$.

\paragraph{Imitation Policy} follows the structure proposed in \cite{yangLyapunovstableNeuralControl2024}, 
as it enforces true control inputs at the equilibrium $\pi_\theta(x^*) = u^*$ and 
satisfies the control input constraints $u_{\min} \leq \pi_\theta(x) \leq u_{\max}$ 
by construction. The neural policy $\pi_\theta$ is visualized in \cref{fig:imitation-policy} and 
formulated as 
\begin{equation}
    \label{eq:policy-arch}
    \pi_\theta(x) = \clamp(
            f_\theta(x) - f_\theta(x^*) + u^*, 
            u_{\min},
            u_{\max}
    ).
\end{equation}
where $\clamp(\cdot,u_{\min},u_{\max})$ is applied element-wise.

\paragraph{Imitation Objective} consists of a state-weighted behavioral cloning loss 
and a dynamics-aware penalty. 
The loss terms are normalized using the diagonal scaling matrices
\begin{equation}
    S_x = \diag(x_{\mathrm{scale}})
    \quad \text{and} \quad
    S_u = \diag(u_{\mathrm{scale}}), 
\end{equation}
where all scaling factors are strictly positive. In addition,
the element-wise rectified linear unit defined in \cref{eq:relu} is utilized. 
The normalizing the distance of the equilibrium is evaluated using the scaled $L_1$-norm: 
\begin{equation}
    \label{eq:state-distance}
    d(x) = \frac{1}{n_x} \left\| S_x^{-1}(x - x^*) \right\|_1.
\end{equation}

To assign appropriate priority during behavioral cloning, a state-dependent weight is defined as
\begin{equation}
    \label{eq:state-weight}
    w_x(x) = w_{\min} + (1 - w_{\min}) \left( \frac{1}{1 + d(x)} \right)^\alpha,
\end{equation}
where $w_{\min} \in (0, 1]$ is the baseline weight for states far from the
equilibrium and $\alpha \geq 0$ controls the decay rate.

\paragraph{State-Weighted Behavioral Cloning Loss} penalizes the normalized Euclidean distance 
between the neural network policy and the expert targets. 
For a mini-batch $\mathcal{B}_\pi \subset \mathcal{D}_\pi$ of size 
$M_\pi = |\mathcal{B}_\pi|$, the behavioral cloning loss is defined as
\begin{equation}
    \label{eq:state-weighted-loss}
    \mathcal{L}_{\mathrm{BC}} = \frac{1}{M_\pi n_u} 
    \sum_{(x, u) \in \mathcal{B}_\pi} w_x(x) \left\| 
        S_u^{-1} \big( \pi_\theta(x) - u \big) 
    \right\|_2^2.
\end{equation}
Here, the state-dependent weight $w_x(x)$ prioritizes approximation accuracy near the equilibrium.


\paragraph{Dynamics-Aware Constraint Loss} applies a one-step penalty for successor states 
that leave the admissible state set $\mathcal{X}_\pi$, defined by lower and upper bounds 
$\underline{x}_\pi$ and $\overline{x}_\pi$. 
For each state $x \in \mathcal{B}_\pi$, the next state is evaluated under 
the neural policy as $x^+ = f_d(x,\pi_\theta(x))$. The loss is defined as
\begin{equation}
    \label{eq:dynamics-aware-loss}
    \mathcal{L}_f = \frac{1}{M_\pi n_x} \sum_{x \in \mathcal{B}_\pi} \left( 
        \left\| 
            \big[ x^+ - \overline{x}_\pi \big]_+ 
        \right\|_1 + \left\| 
            \big[ \underline{x}_\pi - x^+ \big]_+ 
        \right\|_1 
    \right).
\end{equation}
Here, a one-step evaluation outside the admissible set results in a penalty for the current state, 
forcing the learned policy to stay inside the admissible set.


\paragraph{Total Imitation Loss} is the combination of the state-weighted behavioral cloning 
and dynamics-aware constraint losses into a single scalar objective. 
Scaling each term by its respective weight, $\omega_{\mathrm{BC}}$ and $\omega_f$, 
yields the overall policy loss:
\begin{equation}
    \label{eq:combined-policy-loss}
    \mathcal{L}_\pi = 
        \omega_{\mathrm{BC}} \mathcal{L}_{\mathrm{BC}} + 
        \omega_f \mathcal{L}_f.
\end{equation}
