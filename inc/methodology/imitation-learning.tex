\section{Neural Policy Approximation and Imitation Learning}
\label{sec:imitation-learning}


\paragraph{Data Generation} uses the MPC expert policy $\pi_0$ defined in \cref{subsec:closed-loop-stability}.
Closed-loop trajectories $x_{k+1} = f(x_k, \pi_0(x_k))$ are initialized from states sampled across 
the training set 
$\mathcal{X}_\pi : \{ x \in \mathcal{X} \mid \underline{x}_\pi \leq x \leq \overline{x}_\pi \}$. 
Storing the visited states and their corresponding optimal control actions yields the dataset
\begin{equation}
    \label{eq:dataset-definition}
    \mathcal{D}_\pi = \left\{ (x^{(i)}, u^{(i)}) \right\}_{i=1}^{N_{\mathcal{D}}},
\end{equation}
where $x^{(i)} \in \mathcal{X}_\pi$ and $u^{(i)} = \pi_0(x^{(i)})$.


\paragraph{Imitation Policy} follows the structure proposed in \cite{yangLyapunovstableNeuralControl2024}, 
as it enforces true control inputs at the equilibrium $\pi_\theta(x^*) = u^*$ and 
satisfies the control input constraints $\underline{u} \leq \pi_\theta(x) \leq \overline{u}$ 
by construction. The neural policy $\pi_\theta$ is visualized in \cref{fig:imitation-policy} and 
formulated as 
\begin{equation}
    \label{eq:policy-arch}
    \pi_\theta(x) = \clamp(
            h_\theta(x) - h_\theta(x^*) + u^*, 
            \underline{u},
            \overline{u}
    ).
\end{equation}
where $h_\theta : \mathbb{R}^{n_x} \to \mathbb{R}^{n_u}$ is an unconstrained neural network, 
$u^* \in \mathcal{U} $ represents the desired control action at the equilibrium $x^*$, 
and $\clamp(\cdot, \overline{\cdot}, \underline{\cdot})$ acts element-wise to enforce the input constraints.

\paragraph{Imitation Objective} is a multi-objective loss, which consists of two parts: a state-weighted behavioral cloning loss 
and a dynamics-aware penalty. 
To account for different physical units and numerical scales across the state and control vectors, 
the loss terms are normalized using strictly positive diagonal scaling matrices
\begin{equation}
    S_x = \diag(x_{\mathrm{scale}}) \in \mathbb{R}^{n_x \times n_x}
    \quad \text{and} \quad
    S_u = \diag(u_{\mathrm{scale}}) \in \mathbb{R}^{n_u \times n_u}.
\end{equation}
Using $S_x$, the normalized distance of a state $x$ from the equilibrium $x^*$ is quantified with the $\ell_1$-norm as
\begin{equation}
    \label{eq:state-distance}
    d(x) = \frac{1}{n_x} \left\| S_x^{-1}(x - x^*) \right\|_1.
\end{equation}

To assign appropriate priority during behavioral cloning, a state-dependent weight is defined as
\begin{equation}
    \label{eq:state-weight}
    w_x(x) = w_{\min} + (1 - w_{\min}) \left( \frac{1}{1 + d(x)} \right)^\eta,
\end{equation}
where $w_{\min} \in (0, 1)$ is the baseline weight for states far from the
equilibrium and $\eta \geq 0$ controls the decay rate.

\paragraph{State-Weighted Behavioral Cloning Loss} penalizes the squared normalized 
Euclidean distance between the neural network policy and the expert targets. 
For a mini-batch $\mathcal{B}_\pi \subset \mathcal{D}_\pi$ of size 
$M_\pi = |\mathcal{B}_\pi|$, the behavioral cloning loss is defined as
\begin{equation}
    \label{eq:state-weighted-loss}
    \loss_{\mathrm{BC}} = \frac{1}{M_\pi n_u} 
    \sum_{(x, u) \in \mathcal{B}_\pi} w_x(x) \left\| 
        S_u^{-1} \big( \pi_\theta(x) - u \big) 
    \right\|_2^2.
\end{equation}
Here, the state-dependent weight $w_x(x)$ prioritizes approximation accuracy near the equilibrium.


\paragraph{Dynamics-Aware Constraint Loss} applies a one-step penalty for successor states 
that leave the admissible state set $\mathcal{X}_\pi$.
For each state $x \in \mathcal{B}_\pi$, the next state is evaluated under 
the neural policy as $x^+ = f(x,\pi_\theta(x))$. 
Using the element-wise ReLU function defined in \eqref{eq:relu}, the loss is defined as
\begin{equation}
    \label{eq:dynamics-aware-loss}
    \loss_f = \frac{1}{M_\pi n_x} \sum_{x \in \mathcal{B}_\pi} \left( 
        \left\| 
            \big[ x^+ - \overline{x}_\pi \big]_+ 
        \right\|_1 + \left\| 
            \big[ \underline{x}_\pi - x^+ \big]_+ 
        \right\|_1 
    \right).
\end{equation}
Here, a one-step evaluation outside the admissible set results in a penalty for the current state, 
forcing the learned policy to stay inside the admissible set.


\paragraph{Total Imitation Loss} is the combination of the state-weighted behavioral cloning 
and dynamics-aware constraint losses into a single scalar objective. 
Scaling each term by its respective weight, $\omega_{\mathrm{BC}}$ and $\omega_f$, 
yields the overall policy loss:
\begin{equation}
    \label{eq:combined-policy-loss}
    \loss_\pi = 
        \omega_{\mathrm{BC}} \loss_{\mathrm{BC}} + 
        \omega_f \loss_f.
\end{equation}

By minimizing $\loss_\pi$, the policy $\pi_\theta$ learns to mimic the expert policy $\pi_0$, 
while penalizing state constraint violations in $\mathcal{X}_\pi$. 
In combination with the policy architecture in \eqref{eq:policy-arch}, $\pi_\theta$ satisfies the input constraints 
$\underline{u} \leq \pi_\theta(x) \leq \overline{u}$ for all states $x \in \mathbb{R}^{n_x}$ and guarantees exact 
equilibrium control $\pi_\theta(x^*) = u^*$. 
Additionally, the state-dependent weighting in \eqref{eq:state-weight} prioritizes approximation accuracy 
near the equilibrium, which is essential for establishing Lyapunov stability.
Consequently, this imitation learning establishes a highly accurate and safe foundational policy, 
serving as an initialization for the subsequent Lyapunov co-training and verification process.
