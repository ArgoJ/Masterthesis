    \quad
    x + \delta \in \mathcal{X}.
\end{equation}

\subsection{Projected Gradient Descent}
\label{subsec:pgd}

Projected Gradient Descent (PGD) is an iterative gradient-based method for
constructing adversarial examples. Starting from the original input
$x_{\mathrm{adv}} := x$, each iteration modifies the current candidate in a
direction that increases a differentiable objective $\mathcal{J}$:
\begin{equation}
    \label{eq:pgd-update}
    x_{\mathrm{adv}}
    \leftarrow
    \Pi_{\mathcal{C}_{\varepsilon}(x)}
    \left(
        x_{\mathrm{adv}}
        +
        \alpha
        \operatorname{sign}
        \left(
            \nabla_{x_{\mathrm{adv}}}
            \mathcal{J}(\theta, x_{\mathrm{adv}}, y)
        \right)
    \right),
\end{equation}
where $\alpha > 0$ denotes the step size and
\begin{equation}
    \mathcal{C}_{\varepsilon}(x)
    =
    \left\{
        z
        \;\middle|\;
        \|z-x\|_\infty \leq \varepsilon
    \right\}
\end{equation}
defines the admissible $\ell_\infty$-bounded perturbation set.
