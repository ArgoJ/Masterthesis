\begin{equation}
    \label{eq:euler-general}
    x_{k+1} = x_k + T_s f(x_k, u_k).
\end{equation}

\begin{equation}
    \label{eq:rk4-general}
    x_{k+1} = x_k + \frac{T_s}{6} (k_1 + 2k_2 + 2k_3 + k_4),
\end{equation}
where the intermediate slopes $k_1$ through $k_4$ are defined as:
\begin{equation}
    \label{eq:rk4-steps}
    \begin{aligned}
        k_1 &= f(x_k, u_k) \\
        k_2 &= f\left(x_k + \frac{T_s}{2} k_1, u_k\right) \\
        k_3 &= f\left(x_k + \frac{T_s}{2} k_2, u_k\right) \\
        k_4 &= f\left(x_k + T_s k_3, u_k\right).
    \end{aligned}
\end{equation}
