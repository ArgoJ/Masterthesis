\begin{tikzpicture}[>=Latex, font=\small, text=black]

    % --- 1. Schiene / Führung ---
    \draw[thick, black] (-1.8,0) -- (4.8,0);
    \foreach \x in {-1.5,-1.0,...,4.5} {
        \draw[thin, black] (\x,0) -- (\x-0.2,-0.25);
    }

    % Positionsachse p (Schwarz)
    \draw[->, thick, black] (-1.8,-0.6) -- (4.8,-0.6) node[right, font=\footnotesize, text=black] {$p$};
    \draw[thick, black] (0,-0.5) -- (0,-0.7) node[below, font=\scriptsize, text=black] {$0$};

    % --- 2. Wagen (Masse M) ---
    \def\xcart{1.2}
    \def\wcart{2.4}
    \def\hcart{1.1}
    \def\rwheel{0.22}

    % Räder
    \filldraw[fill=white, draw=navy, thick] (\xcart - 0.75, \rwheel) circle (\rwheel);
    \filldraw[fill=white, draw=navy, thick] (\xcart + 0.75, \rwheel) circle (\rwheel);
    \fill[black] (\xcart - 0.75, \rwheel) circle (1.5pt);
    \fill[black] (\xcart + 0.75, \rwheel) circle (1.5pt);

    % Wagenkörper
    \draw[very thick, navy, fill=navy!10, rounded corners=2pt] 
        (\xcart - \wcart/2, 2*\rwheel) rectangle (\xcart + \wcart/2, 2*\rwheel + \hcart);
    \node[font=\bfseries, text=black] at (\xcart - 0.6, 2*\rwheel + \hcart/2) {$M$};

    % Position p Projektion (Schwarz gestrichelt)
    \draw[dashed, black, thick] (\xcart, 2*\rwheel) -- (\xcart, -0.6);
    \draw[thick, black] (\xcart,-0.5) -- (\xcart,-0.7) node[below, font=\scriptsize, text=black] {$p$};

    % Geschwindigkeit v
    \draw[->, very thick, teal] (\xcart + 0.1, 2*\rwheel + 0.3) --++ (1.1, 0) 
        node[right, font=\footnotesize\bfseries, text=black] {$v$};

    % Stellkraft u
    \draw[->, very thick, purple] (\xcart - \wcart/2 - 1.2, 2*\rwheel + \hcart/2) --++ (1.1, 0) 
        node[midway, above, font=\footnotesize\bfseries, text=black] {$u$};

    % --- 3. Invertiertes Pendel (Masse m, Länge l_p) ---
    \coordinate (pivot) at (\xcart, 2*\rwheel + \hcart);
    \def\pendlen{2.8}
    \def\pendangle{24}

    % Gelenk
    \filldraw[fill=white, draw=navy, thick] (pivot) circle (3pt);

    % Vertikale Referenzlinie (Aufrechte Ruhelage \phi = 0, Schwarz gestrichelt)
    \draw[dashed, black, thick] (pivot) --++ (0, 3.2);

    % Pendelstab & Pendelmasse m
    \coordinate (bob) at ($(pivot) + ({sin(\pendangle)*\pendlen}, {cos(\pendangle)*\pendlen})$);
    \draw[very thick, navy] (pivot) -- (bob) 
        node[midway, left=2pt, font=\footnotesize, text=black] {$l_p$};

    \filldraw[fill=amber!25, draw=amber!90!black, very thick] (bob) circle (0.32);
    \node[font=\bfseries, text=black] at (bob) {$m$};

    % Winkel \phi Bogen
    \draw[->, thick, teal] ($(pivot) + (0, 1.6)$) arc (90:90-\pendangle:1.6);
    \node[font=\footnotesize\bfseries, text=black] at ($(pivot) + (0.35, 1.85)$) {$\phi$};

    % Winkelgeschwindigkeit \omega
    \draw[->, thick, teal] ($(bob) + (-0.5, 0.1)$) arc (160:110:0.7) 
        node[above left, font=\footnotesize\bfseries, text=black] {$\omega$};

    % Zustandsvektor-Badge
    \node[draw=black, fill=white, rounded corners=3pt, font=\scriptsize, text=black, anchor=west] at (-1.8, 3.2) {
        $x = \begin{bmatrix} p \\ v \\ \phi \\ \omega \end{bmatrix} \in \mathbb{R}^4$
    };

\end{tikzpicture}
