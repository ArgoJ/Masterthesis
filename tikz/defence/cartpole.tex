\begin{tikzpicture}[>=Latex, font=\small, text=black]

    % --- 1. Schiene / Führung ---
    \draw[thick, black] (-1.8,0) -- (4.8,0);
    \foreach \x in {-1.5,-1.0,...,4.5} {
        \draw[thin, black] (\x,0) -- (\x-0.2,-0.25);
    }

    % Nullreferenz x=0 (gestrichelt)
    \draw[dashed, black!60, thick] (0, -0.9) -- (0, 2.0);
    \node[below, font=\scriptsize, text=black] at (0, -0.9) {$0$};

    % --- 2. Wagen (Masse M) ---
    \def\xcart{1.2}
    \def\wcart{2.4}
    \def\hcart{1.1}
    \def\rwheel{0.22}

    % Pfeil für Position und Geschwindigkeit (p, v) unterhalb der Schiene von x=0 bis zur Wagenmitte
    \draw[dashed, black!60, thick] (\xcart, 2*\rwheel) -- (\xcart, -0.9);
    \draw[->, very thick, teal] (0, -0.65) -- (\xcart, -0.65) 
        node[midway, below=1pt, font=\small\bfseries, text=black] {$p,\,v$};

    % Räder
    \filldraw[fill=white, draw=navy, thick] (\xcart - 0.75, \rwheel) circle (\rwheel);
    \filldraw[fill=white, draw=navy, thick] (\xcart + 0.75, \rwheel) circle (\rwheel);
    \fill[black] (\xcart - 0.75, \rwheel) circle (1.5pt);
    \fill[black] (\xcart + 0.75, \rwheel) circle (1.5pt);

    % Wagenkörper
    \draw[very thick, navy, fill=navy!10, rounded corners=2pt] 
        (\xcart - \wcart/2, 2*\rwheel) rectangle (\xcart + \wcart/2, 2*\rwheel + \hcart);
    \node[font=\bfseries, text=black] at (\xcart, 2*\rwheel + \hcart/2) {$M$};

    % Stellkraft u
    \draw[->, very thick, purple] (\xcart - \wcart/2 - 1.2, 2*\rwheel + \hcart/2) --++ (1.1, 0) 
        node[midway, above, font=\footnotesize\bfseries, text=black] {$u$};

    % --- 3. Invertiertes Pendel (Masse m, Länge l_p) ---
    \coordinate (pivot) at (\xcart, 2*\rwheel + \hcart);
    \def\pendlen{2.8}
    \def\pendangle{24}

    % Gelenk
    \filldraw[fill=white, draw=navy, thick] (pivot) circle (3pt);

    % Vertikale Referenzlinie (Aufrechte Ruhelage \varphi = 0, Schwarz gestrichelt)
    \draw[dashed, black, thick] (pivot) --++ (0, 3.2);

    % Pendelstab & Pendelmasse m
    \coordinate (bob) at ($(pivot) + ({sin(\pendangle)*\pendlen}, {cos(\pendangle)*\pendlen})$);
    \draw[very thick, navy] (pivot) -- (bob) 
        node[midway, right=3pt, font=\footnotesize, text=black] {$l_p$};

    \filldraw[fill=amber!25, draw=amber!90!black, very thick] (bob) circle (0.32);
    \node[font=\bfseries, text=black] at (bob) {$m$};

    % Winkel & Winkelgeschwindigkeit (\varphi, \omega) im Winkelbereich
    \draw[->, very thick, teal] ($(pivot) + (0, 1.9)$) arc (90:90-\pendangle:1.9);
    \node[font=\small\bfseries, text=black] at ($(pivot) + (0.35, 1.45)$) {$\varphi,\,\omega$};

    % Gravitationsbeschleunigung g
    \draw[->, thick, black] (-1.2, 4.3) --++ (0, -0.8) node[midway, left=1pt, font=\small, text=black] {$g$};

\end{tikzpicture}
