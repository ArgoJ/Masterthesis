\begin{tikzpicture}[
    >=stealth, thick,
    node distance=0.6cm and 1.0cm,
    % Styles für Blöcke und Knoten
    blockNN/.style={
        rectangle, draw=teal!90!black, fill=teal!12,
        minimum height=0.8cm, minimum width=1.2cm,
        rounded corners=3pt, font=\small\bfseries
    },
    blockClamp/.style={
        rectangle, draw=amber!90!black, fill=amber!12,
        minimum height=0.8cm, minimum width=2.2cm,
        rounded corners=3pt, font=\small
    },
    sum/.style={
        circle, draw=navy, fill=white,
        inner sep=0pt, minimum size=18pt, font=\small
    },
    val/.style={font=\small}
]

    % --- KNOTEN DEFINITION ---
    % Hauptpfad (Oben)
    \node[val] (x) {$x$};
    \node[blockNN, right=0.8cm of x] (f1) {$h_\theta$};
    \node[sum, right=1.6cm of f1] (sum) {$\Sigma$};
    
    % Sekundärpfad für Gleichgewichtszustand (Unten)
    \node[val, below=1.0cm of x] (xstar) {$x^\star$};
    \node[blockNN, right=0.8cm of xstar] (f2) {$h_\theta$};

    % Gleichgewichtseingang (Oben zentriert über Summe)
    \node[val, above=0.8cm of sum] (ustar) {$u^\star$};

    % Clamp & Ausgang (Rechts)
    \node[blockClamp, right=1.2cm of sum] (clamp) {$\clamp(\cdot, \underline{u}, \overline{u})$};
    \node[val, right=0.8cm of clamp] (out) {$\pi_\theta (x)$};

    % --- VERBINDUNGEN & BESCHRIFTUNGEN ---
    % 1. Pfad f_theta(x)
    \draw[->] (x) -- (f1);
    \draw[->] (f1) -- node[pos=0.88, above=1pt, font=\tiny\bfseries, text=navy] {$+$} (sum);

    % 2. Pfad f_theta(x^\star)
    \draw[->] (xstar) -- (f2);
    \draw[->] (f2) -| node[pos=0.88, right=1pt, font=\tiny\bfseries, text=navy] {$-$} (sum);

    % 3. Pfad u^\star
    \draw[->] (ustar) -- node[pos=0.8, left=1pt, font=\tiny\bfseries, text=navy] {$+$} (sum);

    % 4. Summe zu Clamp und Ausgang
    \draw[->] (sum) -- (clamp);
    \draw[->] (clamp) -- (out);

    % Visualisierung der geteilten Gewichte (Shared Weights)
    \draw[dashed, navy!40, thin] (f1) -- node[right, font=\tiny, text=navy!70] {shared $\theta$} (f2);

\end{tikzpicture}
