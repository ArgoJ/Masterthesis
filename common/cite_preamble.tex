% common/cite_preamble.tex

\ExplSyntaxOn

\clist_new:N \l_smartcite_post_clist
\clist_new:N \l_smartcite_keys_clist
\tl_new:N \l_smartcite_call_tl

\NewDocumentCommand{\smartCite}{m m m}{
  % Clear and populate lists
  \clist_set:Nn \l_smartcite_post_clist {#2}
  \clist_set:Nn \l_smartcite_keys_clist {#3}
  
  % Build the token list: starting with \autocites
  \tl_set:Nn \l_smartcite_call_tl { \autocites }
  
  \int_step_inline:nn { \clist_count:N \l_smartcite_keys_clist }
    {
      \clist_pop:NN \l_smartcite_post_clist \l_tmpa_item
      \clist_pop:NN \l_smartcite_keys_clist \l_tmpb_item
      
      \tl_if_empty:NTF \l_tmpa_item
        { 
          \tl_put_right:Nn \l_smartcite_call_tl { {} }
          \tl_put_right:Nx \l_smartcite_call_tl { { \exp_not:V \l_tmpb_item } }
        }
        { 
          \tl_put_right:Nx \l_smartcite_call_tl { [ #1 ~ \exp_not:V \l_tmpa_item ] }
          \tl_put_right:Nx \l_smartcite_call_tl { { \exp_not:V \l_tmpb_item } }
        }
    }
  
  % Execute the constructed \autocites command
  \tl_use:N \l_smartcite_call_tl
}

\ExplSyntaxOff

% Custom Commands
\newcommand{\citeSec}[2]{\smartCite{Sec.}{#1}{#2}}
\newcommand{\citeDef}[2]{\smartCite{Def.}{#1}{#2}}
\newcommand{\citeAsm}[2]{\smartCite{Assumption}{#1}{#2}}
\newcommand{\citeThm}[2]{\smartCite{Theorem}{#1}{#2}}
\newcommand{\citeEq}[2]{\smartCite{Eq.}{#1}{#2}}
\newcommand{\citeRem}[2]{\smartCite{Remark}{#1}{#2}}
